```python
# Dashboard principal - Nova Tab
main_tabs = st.tabs([
    "📊 IA ESTRATÉGICA",           # 🆕 NOVA TAB - IA GENERATIVA
    "🎯 CAMADA 1: FUNDAMENTOS", 
    "🔬 CAMADA 2: PRESENÇA & CULTURA",
    "🚀 CAMADA 3: FUTUROS",
    "🤖 AGENTE AUTÔNOMO",
    "📈 ML METRICS AVANÇADAS"
])

with main_tabs[0]:  # IA ESTRATÉGICA
    st.markdown("# 🧠 Intelligence Estratégica Baseada em IA")
    st.markdown("*Insights narrativos automáticos sobre sua pesquisa cultural*")
    
    # Área de input de contexto
    col1, col2 = st.columns([2, 1])
    
    with col1:
        st.markdown("### 📝 Contexto do Negócio")
        business_context = st.text_area(
            "Descreva seu cenário de negócio:",
            placeholder="Ex: Queremos entender como jovens da periferia do Rio se relacionam com aplicativos de música...",
            height=100
        )
        
        # Seletor de tipo de análise
        analysis_type = st.selectbox(
            "Tipo de Análise:",
            ["Pesquisa de Mercado", "Lançamento de Produto", "Construção de Marca", "Crise de Reputação", "Expansão"]
        )
        
    with col2:
        st.markdown("### ⚙️ Configurações IA")
        confidence_threshold = st.slider("Nível de Confiança", 0.1, 1.0, 0.7, 0.1)
        max_insights = st.selectbox("Máximo de Insights", [5, 8, 10, 15], 1)
        include_risks = st.checkbox("Incluir Análise de Riscos", True)
    
    # Botão para gerar insights
    if st.button("🧠 Gerar Insights Estratégicos", type="primary"):
        if business_context:
            with st.spinner("🤖 IA analisando contexto cultural..."):
                # Chamar Business Synthesizer
                refined_results = await refine_research_context(
                    f"{analysis_type}: {business_context}",
                    st.session_state.get('current_cultural_data', {})
                )
                
                # Armazenar resultados
                st.session_state['ai_insights'] = refined_results
                st.success("✅ Análise concluída!")
```

        # Criar dados para visualização de rede
        import networkx as nx
        import plotly.graph_objects as go
        
        # Criar grafo
        G = nx.Graph()
        
        # Adicionar nós (círculos culturais principais)
        main_circles = [i.circle_name for i in results['insights'][:5]]
        for circle in main_circles:
            G.add_node(circle.replace('_', ' ').title())
        
        # Adicionar arestas (pontes)
        for bridge in results['strategy'].cultural_bridges:
            # Simplificado - conectar círculos relacionados
            if len(main_circles) >= 2:
                G.add_edge(main_circles[0].replace('_', ' ').title(), 
                          main_circles[1].replace('_', ' ').title())
        
        # Layout do grafo
        pos = nx.spring_layout(G)
        
        # Plotly visualization
        edge_x, edge_y = [], []
        for edge in G.edges():
            x0, y0 = pos[edge[0]]
            x1, y1 = pos[edge[1]]
            edge_x.extend([x0, x1, None])
            edge_y.extend([y0, y1, None])
        
        node_x = [pos[node][0] for node in G.nodes()]
        node_y = [pos[node][1] for node in G.nodes()]
        
        fig = go.Figure(data=[
            go.Scatter(x=edge_x, y=edge_y, mode='lines', line=dict(width=2, color='#888')),
            go.Scatter(x=node_x, y=node_y, mode='markers+text', 
                      text=list(G.nodes()), textposition="middle center",
                      marker=dict(size=20, color='lightblue'))
        ])
        
        fig.update_layout(title="Rede de Conexões Culturais", showlegend=False)
        st.plotly_chart(fig, use_container_width=True)
